\section{Theoretical Analysis}
\label{sec:theoretical_analysis}

We analyze benign trajectory stability, multi-resolution reference separation, harmful influence, stationarity, and monitoring recovery. Full proofs appear in Appendix~\ref{app:full_theoretical_proofs}. Partition the nonempty valid cohort as $\mathcal{A}_t=\mathcal{G}_t\mathbin{\dot\cup}\mathcal{B}_t$ into benign and harmful participants, unknown to the defense. For the influence results, assume $q_i^t\geq q_{\mathrm{ben}}$ for $c_i\in\mathcal{G}_t$ and $q_i^t\leq q_{\mathrm{bad}}$ for $c_i\in\mathcal{B}_t$, with $0<q_{\mathrm{bad}}\leq q_{\mathrm{ben}}<1$. Score separation is assumed, not guaranteed by self-supervised calibration; clipping merely places scores strictly between zero and one. Empty-cohort rounds leave the model unchanged and are excluded from effective-round indexing. Convergence concerns a stationary reference objective $F$; time-varying $F^t$ requires additional control of objective drift.

\subsection{Trajectory Stability and Temporal References}

Online encoding introduces benign input variation and encoder-parameter drift. Let $t_{i,n}$ be client $c_i$'s $n$th protocol-valid participation round, with $\mathbf{x}_{i,n}=\mathbf{x}_i^{t_{i,n}}$, $\phi_{i,n}=\phi^{t_{i,n}}$, $\mathbf{h}_{i,n}=\Phi_{\phi_{i,n}}(\mathbf{h}_{i,n-1},\mathbf{x}_{i,n})$, and $\mathbf{z}_{i,n}=\Psi_{\phi_{i,n}}(\mathbf{h}_{i,n})$. Here $\Phi$ is the recurrent transition and $\Psi$ the latent head; historical hidden states are detached (Section~\ref{subsec:trajectory_encoding}).

\begin{proposition}[Benign trajectory stability under online encoding]
\label{prop:benign_representation_stability}
Suppose the recurrent transition and latent head satisfy, for $0\leq\chi<1$ and finite positive $L_x,L_\phi,L_z,L_\psi$,
\begin{equation}
\begin{aligned}
&\|\Phi_{\phi}(\mathbf{h},\mathbf{x})-\Phi_{\phi'}(\mathbf{h}',\mathbf{x}')\|_2\\
&\quad\leq\chi\|\mathbf{h}-\mathbf{h}'\|_2+L_x\|\mathbf{x}-\mathbf{x}'\|_2\\
&\qquad+L_\phi\|\phi-\phi'\|_2,\\
&\|\Psi_{\phi}(\mathbf{h})-\Psi_{\phi'}(\mathbf{h}')\|_2\\
&\quad\leq L_z\|\mathbf{h}-\mathbf{h}'\|_2+L_\psi\|\phi-\phi'\|_2.
\end{aligned}
\label{eq:theory_encoder_regularities}
\end{equation}
Here $D_\phi$ captures cumulative parameter changes between consecutive valid participations. If a benign client satisfies $\|\mathbf{x}_{i,n}-\mathbf{x}_{i,n-1}\|_2\leq D_{\mathrm{ben}}$ and $\|\phi_{i,n}-\phi_{i,n-1}\|_2\leq D_\phi$ between successive valid participations, then, with $C_h=L_xD_{\mathrm{ben}}+L_\phi D_\phi$,
\begin{equation}
\begin{aligned}
\limsup_{n\to\infty}\|\mathbf{z}_{i,n}-\mathbf{z}_{i,n-1}\|_2&\leq B_z,\\
B_z&:=\frac{L_zC_h}{1-\chi}+L_\psi D_\phi.
\end{aligned}
\label{eq:latent_stability_main}
\end{equation}
If additionally $\|\mathbf{z}_{i,n}\|_2\geq m_z\geq\varepsilon_{\cos}>0$ for all sufficiently large $n$, then the stabilized cosine similarity used in Section~\ref{subsec:method_multi_resolution} satisfies
\begin{equation}
\limsup_{n\to\infty}\big[1-\cos_{\varepsilon}(\mathbf{z}_{i,n},\mathbf{z}_{i,n-1})\big]\leq\frac{B_z^2}{2m_z^2}.
\label{eq:latent_cosine_stability}
\end{equation}
\end{proposition}

\noindent\textit{Proof sketch.} For $d_n=\|\mathbf{h}_{i,n}-\mathbf{h}_{i,n-1}\|_2$, the assumed regularity yields $d_n\leq\chi d_{n-1}+C_h$. Iterating this contraction and applying the projection-head bound establishes Eq.~\eqref{eq:latent_stability_main}. With nonvanishing latent norms, stabilized cosine equals ordinary cosine and Euclidean distance bounds angular disagreement, giving Eq.~\eqref{eq:latent_cosine_stability}. Contraction, bounded encoder drift, and nonvanishing latent norms (for cosine) are assumptions, not generic GRU guarantees.

\begin{lemma}[Fast--slow reference separation under finite drift]
\label{lem:fast_slow_separation}
Let $s_i^n=\langle\mathbf{z}_{i,n},\mathbf{v}\rangle$ denote the scalar projection of the client's latent representation along a unit direction $\mathbf{v}$. Suppose $s_i^n=s_i^0+\nu_{\mathrm{drift}}n$ for $0\leq n\leq N$, where $N\geq1$ and $\nu_{\mathrm{drift}}>0$. Initialize both scalar EMA anchors at $s_i^0$, with $1>\gamma_{\mathrm{fast}}>\gamma_{\mathrm{slow}}>0$ and $r_{\mathrm{fast}}=1-\gamma_{\mathrm{fast}}$, $r_{\mathrm{slow}}=1-\gamma_{\mathrm{slow}}$. For $1\leq n\leq N$,
\begin{equation}
\bar{s}_{i,\mathrm{fast}}^n-\bar{s}_{i,\mathrm{slow}}^n=\nu_{\mathrm{drift}}\sum_{k=1}^{n}(r_{\mathrm{slow}}^k-r_{\mathrm{fast}}^k)>0.
\label{eq:fast_slow_main}
\end{equation}
\end{lemma}

\noindent\textit{Proof sketch.} Under the stated linear drift, an EMA with coefficient $\gamma$ has a lag after $n$ observations of $\nu_{\mathrm{drift}}\sum_{k=1}^{n}(1-\gamma)^k$; subtracting the lags proves Eq.~\eqref{eq:fast_slow_main}. This shows finite-window scalar separation, not guaranteed detection through cosine-based $\Delta_{\mathrm{MR}}^t(i)$: different anchors may still have equal cosine similarity to the current representation. Features at participation $n$ use anchors stored after participation $n-1$, making the separation available to later comparisons.

\subsection{Harmful Influence and Convergence}

Let $E_{\mathrm{bad}}^t=\sum_{c_i\in\mathcal{B}_t}\alpha_i^t\Delta w_i^t$ be the harmful aggregate component, with exactly normalized weights from Eq.~\eqref{eq:interaction_aware_weight}.

\begin{theorem}[Bounded harmful aggregation influence]
\label{thm:bounded_harmful_influence}
Assume $\|\Delta w_i^t\|_2\leq G_{\mathrm{bad}}<\infty$ for each $c_i\in\mathcal{B}_t$ and the stated score separation. Let $s(q)=q^2/(1-q)$, $U_t=|\mathcal{B}_t|s(q_{\mathrm{bad}})$, and $V_t=|\mathcal{G}_t|s(q_{\mathrm{ben}})$. For each nonempty valid cohort,
\begin{equation}
\|E_{\mathrm{bad}}^t\|_2\leq A_t:=G_{\mathrm{bad}}\frac{U_t}{U_t+V_t}.
\label{eq:harmful_bound_main}
\end{equation}
\end{theorem}

\noindent\textit{Proof sketch.} Since $q_i^t=\sigma(\widetilde r_i^t)$, the unnormalized weight is $q_i^t\exp(\widetilde r_i^t)=s(q_i^t)$, increasing on $(0,1)$. Score separation bounds harmful weight from above and benign weight from below. Combining the resulting fraction with bounded harmful-update norms proves the claim. Neither separation nor raw-update clipping follows from calibration; normalizing encoder inputs likewise does not bound raw aggregation updates. If every protocol-valid participant is harmful, their aggregation weights sum to one even when each reliability score is low.

For convergence, define $G_{\mathrm{ben}}^t=\sum_{c_i\in\mathcal{G}_t}\alpha_i^t\Delta w_i^t$. Let $\mathcal{F}_t$ denote the server history before the random events of effective round $t$, with $w^t$ measurable under $\mathcal{F}_t$. The history includes retained client states and prior participation. For $a,b_1,b_2>0$ and $\sigma^2\geq0$, assume
\begin{equation}
\begin{aligned}
\mathbb{E}[\langle\nabla F(w^t),G_{\mathrm{ben}}^t\rangle\mid\mathcal{F}_t]&\leq-a\|\nabla F(w^t)\|_2^2,\\
\mathbb{E}[\|G_{\mathrm{ben}}^t\|_2^2\mid\mathcal{F}_t]&\leq b_1\|\nabla F(w^t)\|_2^2+b_2\sigma^2.
\end{aligned}
\label{eq:benign_aggregate_conditions}
\end{equation}
These expected-descent and second-moment assumptions concern the reliability-weighted benign aggregate under random participation; score separation alone does not ensure descent.

\begin{theorem}[Average stationarity under bounded harmful influence]
\label{thm:trajrep_stationarity}
Let $F$ be $L$-smooth and bounded below by $F_{\inf}$. Suppose Eq.~\eqref{eq:benign_aggregate_conditions} holds and the harmful-influence bound of Theorem~\ref{thm:bounded_harmful_influence} applies at each effective round, with constant $0<\eta\leq a/(4Lb_1)$. Define $\overline{A^2}_T=T^{-1}\sum_{t=0}^{T-1}\mathbb{E}[A_t^2]$. After $T$ nonempty-cohort rounds,
\begin{equation}
\begin{aligned}
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla F(w^t)\|_2^2\leq{}&\frac{4[F(w^0)-F_{\inf}]}{\eta aT}\\
&+\frac{4L\eta b_2}{a}\sigma^2\\
&+\frac{4}{a}\left(\frac{1}{2a}+L\eta\right)\overline{A^2}_T.
\end{aligned}
\label{eq:stationarity_main}
\end{equation}
\end{theorem}

\noindent\textit{Proof sketch.} Apply $L$-smoothness to $w^{t+1}=w^t+\eta(G_{\mathrm{ben}}^t+E_{\mathrm{bad}}^t)$, use Eq.~\eqref{eq:benign_aggregate_conditions}, and bound the harmful cross term by Young's inequality. Under the specified step-size condition, the resulting bound retains a descent coefficient of at least $(\eta a/4)\|\nabla F(w^t)\|_2^2$; taking expectations and telescoping proves Eq.~\eqref{eq:stationarity_main}.

The terms capture an optimization transient, stochastic variance, and residual squared harmful influence. Better score separation reduces the residual with other quantities fixed. This theorem guarantees average stationarity relative to stationary $F$; it does not establish exact convergence, tracking of nonstationary objectives, or robustness without the stated assumptions.

\subsection{Finite-Participation Monitoring Recovery}

\begin{proposition}[Finite-participation recovery]
\label{prop:finite_time_recovery}
Suppose a monitored benign client satisfies $q_i^k\geq\delta^k+\varepsilon_q$ for $N_w$ consecutive protocol-valid participations, with $\varepsilon_q>0$. If
\begin{equation}
\sigma(\kappa_{\mathrm{rec}}\varepsilon_q)>\theta_{\mathrm{rec}},
\label{eq:theory_recovery_condition}
\end{equation}
the client returns to the normal monitoring state no later than the end of those $N_w$ participations.
\end{proposition}

\noindent\textit{Proof sketch.} Sigmoid monotonicity gives $\rho_i^k\geq\sigma(\kappa_{\mathrm{rec}}\varepsilon_q)>\theta_{\mathrm{rec}}$ throughout the window. The mean over the most recent $N_w$ valid participations therefore exceeds the recovery threshold in Eq.~\eqref{eq:method_recovery}. Recovery changes monitoring status only, not reliability scores, client eligibility, aggregation weights, or the global model.
